% Helper to escape LaTeX special chars
{% macro latex_escape(text) -%}
{{ text
    | replace('&', '\&')
    | replace('%', '\%')
    | replace('$', '\$')
    | replace('#', '\#')
    | replace('_', '\_')
    | replace('{', '\{')
    | replace('}', '\}')
    | replace('~', '\textasciitilde{}')
    | replace('^', '\^{}')
    | replace('\\', '\textbackslash{}') }}
{%- endmacro %}
